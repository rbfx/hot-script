# Rbfx.Script.Core: the shared script logic. It has no plugin entry point.
# The game plugin and the editor plugin link it and give it a script source.

file(GLOB RBFX_SCRIPT_CORE_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/*.h ${CMAKE_CURRENT_SOURCE_DIR}/*.cpp)
add_library(Rbfx.Script.Core STATIC ${RBFX_SCRIPT_CORE_SOURCES})
target_link_libraries(Rbfx.Script.Core PUBLIC Urho3D hot::backend)
target_include_directories(Rbfx.Script.Core PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/..>
    $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/generated>)
set_target_properties(Rbfx.Script.Core PROPERTIES POSITION_INDEPENDENT_CODE ON)

# The scripts include the engine headers, so the script compile needs the engine include
# folders and the engine definitions. The core target has them as transitive usage
# requirements of Urho3D. The generated file holds the values for the run time.
set(_rbfx_script_flags_target Rbfx.Script.Core)
set(_rbfx_script_inc "$<TARGET_PROPERTY:${_rbfx_script_flags_target},INCLUDE_DIRECTORIES>")
set(_rbfx_script_def "$<TARGET_PROPERTY:${_rbfx_script_flags_target},COMPILE_DEFINITIONS>")
# Remove the definitions that are private to the core and to Hot++.
set(_rbfx_script_def "$<FILTER:${_rbfx_script_def},EXCLUDE,^(HOT_|RBFX_SCRIPT_)>")
file(GENERATE
    OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/generated/Rbfx.Script.Core/ScriptBuildConfig.inl
    CONTENT "// Generated by CMake. Do not edit.
// The engine include folders and definitions for the run-time compile of the scripts.
static const char* const kEngineIncludeDirs[] = {
$<$<BOOL:${_rbfx_script_inc}>:    \"$<JOIN:$<REMOVE_DUPLICATES:${_rbfx_script_inc}>,\",\n    \">\",
>    nullptr};
static const char* const kEngineDefines[] = {
$<$<BOOL:${_rbfx_script_def}>:    R\"rbfxdef($<JOIN:$<REMOVE_DUPLICATES:${_rbfx_script_def}>,)rbfxdef\",\n    R\"rbfxdef(>)rbfxdef\",
>    nullptr};
")
